# React Integration

The `fuzzyfilter-react` package provides a React hook for building filter interfaces with FuzzyFilter.

## Installation

```bash
npm install @jasperhino/fuzzyfilter fuzzyfilter-react
```

## useFuzzyFilter Hook

The main export is the `useFuzzyFilter` hook which provides reactive state management for filter suggestions.

### Basic Usage

```tsx
import { useFuzzyFilter } from "fuzzyfilter-react";
import { createFuzzyFilter, columnId } from "@jasperhino/fuzzyfilter";

// Create and configure the filter (outside component or with useMemo)
const filter = createFuzzyFilter();
filter.setSchema({
  columns: [
    { id: columnId("status"), name: "Status", type: "enum", values: ["Open", "Closed"] },
    { id: columnId("priority"), name: "Priority", type: "number" },
  ],
});
filter.indexData(myData);

function FilterBox() {
  const {
    query,
    setQuery,
    suggestions,
    selectedIndex,
    selectSuggestion,
    applySuggestion,
    navigateSuggestions,
    isLoading,
  } = useFuzzyFilter(filter);

  return (
    <div>
      <input
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        onKeyDown={(e) => {
          if (e.key === "ArrowDown") navigateSuggestions("down");
          if (e.key === "ArrowUp") navigateSuggestions("up");
          if (e.key === "Enter") applySuggestion();
        }}
        placeholder="Type to filter..."
      />
      {isLoading && <span>Loading...</span>}
      <ul>
        {suggestions.map((suggestion, index) => (
          <li
            key={suggestion.id}
            onClick={() => {
              selectSuggestion(index);
              applySuggestion();
            }}
            className={index === selectedIndex ? "selected" : ""}
          >
            {suggestion.label}
            <span className="count">({suggestion.resultCount})</span>
          </li>
        ))}
      </ul>
    </div>
  );
}
```

### With Keyboard Navigation

```tsx
function FilterBox() {
  const {
    query,
    setQuery,
    suggestions,
    selectedIndex,
    navigateSuggestions,
    applySuggestion,
  } = useFuzzyFilter(filter);

  const handleKeyDown = (e: React.KeyboardEvent) => {
    switch (e.key) {
      case "ArrowDown":
        e.preventDefault();
        navigateSuggestions("down");
        break;
      case "ArrowUp":
        e.preventDefault();
        navigateSuggestions("up");
        break;
      case "Enter":
        e.preventDefault();
        applySuggestion();
        break;
    }
  };

  return (
    <input
      value={query}
      onChange={(e) => setQuery(e.target.value)}
      onKeyDown={handleKeyDown}
    />
  );
}
```

## Hook Options

```typescript
interface UseFuzzyFilterOptions {
  /** Initial query string */
  initialQuery?: string;

  /** Debounce delay in milliseconds (default: 150) */
  debounceMs?: number;

  /** Callback when a suggestion is applied */
  onApply?: (suggestion: FilterSuggestion) => void;

  /** Filter context (already-applied filters for stacked counts) */
  filterContext?: CompiledFilter[];
}
```

### Example with Options

```tsx
const { query, suggestions } = useFuzzyFilter(filter, {
  initialQuery: "status eq",
  debounceMs: 200,
  onApply: (suggestion) => {
    console.log("Applied:", suggestion.label);
  },
});
```

### With Stacked Filter Counts

Use `filterContext` to show counts that reflect already-applied filters:

```tsx
const [appliedFilters, setAppliedFilters] = useState<CompiledFilter[]>([]);

const { query, suggestions, applySuggestion } = useFuzzyFilter(filter, {
  filterContext: appliedFilters,
  onApply: (suggestion) => {
    const compiled = filter.compileFilter(
      suggestion.column.id,
      suggestion.operator,
      suggestion.value?.value
    );
    if (compiled) setAppliedFilters((prev) => [...prev, compiled]);
  },
});
// Suggestion counts now reflect the subset of data matching appliedFilters
```

## Return Value

The hook returns an object with the following properties:

| Property | Type | Description |
|----------|------|-------------|
| `query` | `string` | Current query string |
| `suggestionsQuery` | `string` | The query used to generate current suggestions (after debounce) |
| `setQuery` | `(query: string) => void` | Update the query |
| `suggestions` | `FilterSuggestion[]` | Current suggestions |
| `selectedIndex` | `number` | Currently selected suggestion index |
| `selectedSuggestion` | `FilterSuggestion \| null` | The currently selected suggestion |
| `selectSuggestion` | `(index: number) => void` | Select a suggestion by index |
| `navigateSuggestions` | `(direction: "up" \| "down") => void` | Navigate suggestions with wrap-around |
| `applySuggestion` | `() => void` | Apply the selected suggestion |
| `reset` | `() => void` | Reset all state to initial values |
| `isLoading` | `boolean` | Whether suggestions are being generated |
| `error` | `Error \| null` | Error state if suggestion fetch failed |

### Indexing & Data Mutation

| Property | Type | Description |
|----------|------|-------------|
| `isIndexing` | `boolean` | True while async indexing is in progress |
| `indexProgress` | `IndexProgress \| null` | Progress of current indexing operation |
| `telemetryEvents` | `WideEvent[]` | Telemetry events (when `benchmark: true`) |
| `addRow` | `(row: Record<string, unknown>) => void` | Add a row and re-index |
| `deleteRow` | `(index: number) => void` | Delete a row by index and re-index |
| `reindex` | `() => Promise<void>` | Trigger async re-indexing |
| `getData` | `() => Array<Record<string, unknown>>` | Get current data array |

### Example with Data Mutations

```tsx
const { isIndexing, indexProgress, addRow, deleteRow } = useFuzzyFilter(filter);

// Show indexing indicator
if (isIndexing) {
  return <div>Indexing... {indexProgress?.percentage}%</div>;
}

// Add a new row
const handleAdd = () => {
  addRow({ status: "Open", assignee: "New Person" });
};

// Delete row at index 5
const handleDelete = () => {
  deleteRow(5);
};
```

## Best Practices

### 1. Create Filter Outside Component

Create the FuzzyFilter instance outside your component or use `useMemo` to avoid recreating it on every render:

```tsx
// ✅ Good: Outside component
const filter = createFuzzyFilter();

function App() {
  const { suggestions } = useFuzzyFilter(filter);
}

// ✅ Good: With useMemo
function App({ schema, data }) {
  const filter = useMemo(() => {
    const f = createFuzzyFilter();
    f.setSchema(schema);
    f.indexData(data);
    return f;
  }, [schema, data]);

  const { suggestions } = useFuzzyFilter(filter);
}
```

### 2. Update Data When It Changes

Use the built-in `addRow` and `deleteRow` methods, or re-index when data changes:

```tsx
const { reindex } = useFuzzyFilter(filter);

useEffect(() => {
  filter.indexData(data);
  // Or use: reindex();
}, [data]);
```

### 3. Handle Loading States

```tsx
const { isLoading, suggestions } = useFuzzyFilter(filter);

return (
  <div>
    {isLoading ? <Spinner /> : <SuggestionList suggestions={suggestions} />}
  </div>
);
```
