# Vue Integration

The `fuzzyfilter-vue` package provides a Vue 3 composable for building filter interfaces with FuzzyFilter.

## Installation

```bash
npm install @jasperhino/fuzzyfilter fuzzyfilter-vue
```

## useFuzzyFilter Composable

The main export is the `useFuzzyFilter` composable which provides reactive state management using Vue's Composition API.

### Basic Usage

```vue
<script setup lang="ts">
import { useFuzzyFilter } from "fuzzyfilter-vue";
import { createFuzzyFilter, columnId } from "@jasperhino/fuzzyfilter";

// Create and configure the filter
const filter = createFuzzyFilter();
filter.setSchema({
  columns: [
    { id: "status", name: "Status", type: "enum", values: ["Open", "Closed"] },
    { id: "priority", name: "Priority", type: "number" },
  ],
});
filter.indexData(myData);

const {
  query,
  suggestions,
  selectedIndex,
  navigateSuggestions,
  applySuggestion,
  isLoading,
} = useFuzzyFilter(filter);
</script>

<template>
  <div>
    <input
      v-model="query"
      placeholder="Type to filter..."
      @keydown.down.prevent="navigateSuggestions('down')"
      @keydown.up.prevent="navigateSuggestions('up')"
      @keydown.enter="applySuggestion"
    />
    <span v-if="isLoading">Loading...</span>
    <ul>
      <li
        v-for="(suggestion, index) in suggestions"
        :key="suggestion.id"
        :class="{ selected: index === selectedIndex }"
        @click="selectSuggestion(index); applySuggestion()"
      >
        {{ suggestion.label }}
        <span class="count">({{ suggestion.resultCount }})</span>
      </li>
    </ul>
  </div>
</template>
```

### With Keyboard Navigation

```vue
<script setup lang="ts">
import { useFuzzyFilter } from "fuzzyfilter-vue";

const { query, suggestions, selectedIndex, navigateSuggestions, applySuggestion } = useFuzzyFilter(filter);

function handleKeyDown(e: KeyboardEvent) {
  switch (e.key) {
    case "ArrowDown":
      e.preventDefault();
      navigateSuggestions("down");
      break;
    case "ArrowUp":
      e.preventDefault();
      navigateSuggestions("up");
      break;
    case "Enter":
      e.preventDefault();
      applySuggestion();
      break;
  }
}
</script>

<template>
  <input v-model="query" @keydown="handleKeyDown" />
</template>
```

## Composable Options

```typescript
interface UseFuzzyFilterOptions {
  /** Initial query string */
  initialQuery?: string;

  /** Debounce delay in milliseconds (default: 150) */
  debounceMs?: number;

  /** Callback when a suggestion is applied */
  onApply?: (suggestion: FilterSuggestion) => void;

  /** Reactive filter context (already-applied filters for stacked counts) */
  filterContext?: Ref<CompiledFilter[]>;
}
```

### Example with Options

```vue
<script setup lang="ts">
const { query, suggestions } = useFuzzyFilter(filter, {
  initialQuery: "status eq",
  debounceMs: 200,
  onApply: (suggestion) => {
    console.log("Applied:", suggestion.label);
  },
});
</script>
```

### With Stacked Filter Counts

Use `filterContext` to show counts that reflect already-applied filters:

```vue
<script setup lang="ts">
import { ref } from "vue";
import type { CompiledFilter } from "@jasperhino/fuzzyfilter";

const appliedFilters = ref<CompiledFilter[]>([]);

const { query, suggestions } = useFuzzyFilter(filter, {
  filterContext: appliedFilters,
  onApply: (suggestion) => {
    const compiled = filter.compileFilter(
      suggestion.column.id,
      suggestion.operator,
      suggestion.value?.value
    );
    if (compiled) appliedFilters.value.push(compiled);
  },
});
// Suggestion counts now reflect the subset of data matching appliedFilters
</script>
```

## Return Value

The composable returns an object with the following reactive properties:

| Property | Type | Description |
|----------|------|-------------|
| `query` | `Ref<string>` | Current query string (v-model compatible) |
| `suggestionsQuery` | `Ref<string>` | The query used to generate current suggestions (after debounce) |
| `suggestions` | `Ref<FilterSuggestion[]>` | Current suggestions |
| `selectedIndex` | `Ref<number>` | Currently selected suggestion index |
| `selectedSuggestion` | `ComputedRef<FilterSuggestion \| null>` | The currently selected suggestion |
| `setQuery` | `(query: string) => void` | Set the query programmatically |
| `selectSuggestion` | `(index: number) => void` | Select a suggestion by index |
| `navigateSuggestions` | `(direction: "up" \| "down") => void` | Navigate suggestions with wrap-around |
| `applySuggestion` | `() => void` | Apply the selected suggestion |
| `reset` | `() => void` | Reset all state to initial values |
| `isLoading` | `Ref<boolean>` | Whether suggestions are being generated |
| `error` | `Ref<Error \| null>` | Error state if suggestion fetch failed |

### Indexing & Data Mutation

| Property | Type | Description |
|----------|------|-------------|
| `isIndexing` | `Ref<boolean>` | True while async indexing is in progress |
| `indexProgress` | `Ref<IndexProgress \| null>` | Progress of current indexing operation |
| `telemetryEvents` | `Ref<WideEvent[]>` | Telemetry events (when `benchmark: true`) |
| `addRow` | `(row: Record<string, unknown>) => void` | Add a row and re-index |
| `deleteRow` | `(index: number) => void` | Delete a row by index and re-index |
| `reindex` | `() => Promise<void>` | Trigger async re-indexing |
| `getData` | `() => Array<Record<string, unknown>>` | Get current data array |

### Example with Data Mutations

```vue
<script setup lang="ts">
const { isIndexing, indexProgress, addRow, deleteRow } = useFuzzyFilter(filter);

// Add a new row
function handleAdd() {
  addRow({ status: "Open", assignee: "New Person" });
}

// Delete row at index 5
function handleDelete() {
  deleteRow(5);
}
</script>

<template>
  <div v-if="isIndexing">Indexing... {{ indexProgress?.percentage }}%</div>
  <button @click="handleAdd">Add Row</button>
</template>
```

## Best Practices

### 1. Create Filter in Setup

Create the FuzzyFilter instance in your setup function or as a singleton:

```vue
<script setup lang="ts">
import { computed, watch } from "vue";
import { useFuzzyFilter } from "fuzzyfilter-vue";
import { createFuzzyFilter, columnId } from "@jasperhino/fuzzyfilter";

const props = defineProps<{
  data: Record<string, any>[];
}>();

// Create filter once
const filter = createFuzzyFilter();
filter.setSchema({
  columns: [
    { id: "status", name: "Status", type: "enum", values: ["Open", "Closed"] },
  ],
});

// Re-index when data changes
watch(() => props.data, (newData) => {
  filter.indexData(newData);
}, { immediate: true });

const { query, suggestions } = useFuzzyFilter(filter);
</script>
```

### 2. Use with vue-i18n

```vue
<script setup lang="ts">
import { useI18n } from "vue-i18n";
import { createVueI18nProvider } from "@jasperhino/fuzzyfilter/i18n/adapters/vue-i18n";

const i18n = useI18n();
const provider = createVueI18nProvider(i18n);

const filter = createFuzzyFilter({ i18nProvider: provider });
</script>
```

### 3. Handle Loading States

```vue
<template>
  <div>
    <input v-model="query" />
    <div v-if="isLoading">
      <Spinner />
    </div>
    <SuggestionList v-else :suggestions="suggestions" />
  </div>
</template>
```

### 4. Provide/Inject Pattern

For sharing a filter across components:

```vue
<!-- Parent component -->
<script setup lang="ts">
import { provide } from "vue";
import { createFuzzyFilter } from "@jasperhino/fuzzyfilter";

const filter = createFuzzyFilter();
// ... configure filter

provide("fuzzyFilter", filter);
</script>

<!-- Child component -->
<script setup lang="ts">
import { inject } from "vue";
import { useFuzzyFilter } from "fuzzyfilter-vue";
import type { FuzzyFilter } from "@jasperhino/fuzzyfilter";

const filter = inject<FuzzyFilter>("fuzzyFilter")!;
const { query, suggestions } = useFuzzyFilter(filter);
</script>
```
